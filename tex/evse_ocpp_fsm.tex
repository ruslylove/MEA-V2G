\documentclass[tikz, border=10pt]{standalone}
\usetikzlibrary{automata, positioning, arrows, calc}

\begin{document}

\begin{tikzpicture}[>=stealth', shorten >=1pt, auto, node distance=3cm, semithick]
  \tikzstyle{every state}=[fill=white, draw=black, text=black, font=\small, minimum size=1.5cm]
  \tikzstyle{mainloop}=[draw=blue, ultra thick, text=blue]
  \tikzset{mainnode/.style={fill=blue!5, draw=blue!80, thick}}

  % States
  \node[initial, state] (Unavailable) {Unavailable};
  \node[state, mainnode] (Available) [right=of Unavailable] {Available};
  \node[state] (Reserved) [above=of Available] {Reserved};
  \node[state, mainnode] (Preparing) [right=of Available] {Preparing};
  \node[state, mainnode] (Charging) [right=of Preparing] {Charging};
  \node[state] (SuspendedEV) [above=of Charging] {SuspendedEV};
  \node[state] (SuspendedEVSE) [below=1.5cm of Charging] {SuspendedEVSE};
  \node[state, mainnode] (Finishing) [right=of Charging] {Finishing};
  \node[state] (Faulted) [below=of Available, fill=red!10] {Faulted};

  % Transitions
  \path[->] 
    (Unavailable) edge [] node {Init} (Available)
    (Available) edge [bend left] node {Inoperative} (Unavailable)
    (Unavailable) edge [bend left] node {Operative} (Available)
    (Available) edge [bend left] node {Reserve} (Reserved)
    (Reserved) edge [bend left] node {Expiry/Cancel} (Available)
    (Available) edge [mainloop] node {Connected+Auth} (Preparing)
    (Reserved) edge [bend left] node {Connected+Auth} (Preparing)
    
    (Preparing) edge [mainloop] node {HV Mode} (Charging)
    (Charging) edge [bend left] node {$T=0$} (SuspendedEV)
    (SuspendedEV) edge [bend left] node {$T>0 \ \& \ L>0$} (Charging)
    
    (Charging) edge [bend left] node {$L=0$} (SuspendedEVSE)
    (SuspendedEVSE) edge [bend left] node {$L>0 \ \& \ T>0$} (Charging)
    
    (Charging) edge [mainloop] node {Stopped} (Finishing)
    (SuspendedEV) edge [bend left] node {Timeout} (Finishing)
    (SuspendedEVSE) edge [bend right] node {Timeout} (Finishing)
    
    (Finishing) edge [bend left, mainloop] node {Unplugged} (Available)
    (Finishing) edge [bend left=80, dashed] node [left] {Unplugged + Scheduled} (Unavailable)
    
    % Fault transitions
    (Preparing) edge [bend left] node {Error} (Faulted)
    (Charging) edge [bend left] node {Error} (Faulted)
    (Faulted) edge [bend left] node[swap] {Reset} (Available)
    (Faulted) edge [bend left] node[swap] {Inoperative} (Unavailable);

\end{tikzpicture}

\end{document}
